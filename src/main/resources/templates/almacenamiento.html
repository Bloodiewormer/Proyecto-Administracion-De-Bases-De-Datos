Para entornos de producción donde se requiere registrar el crecimiento histórico directamente en el motor Oracle sin depender exclusivamente del sondeo externo, se documenta el mecanismo estándar de administración mediante una tabla de historial y un job programado con DBMS_SCHEDULER: 

-- 1. Tabla para almacenar snapshots periódicos de almacenamiento
CREATE TABLE HISTORIAL_CRECIMIENTO_BD (
    ID NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    FECHA TIMESTAMP DEFAULT SYSTIMESTAMP,
    BYTES_DATAFILES NUMBER,
    BYTES_TEMPFILES NUMBER,
    BYTES_REDO NUMBER DEFAULT 0,
    BYTES_USADOS NUMBER,
    BYTES_LIBRES NUMBER
);

-- 2. Procedimiento administrativo para recolectar métricas
CREATE OR REPLACE PROCEDURE REGISTRAR_CRECIMIENTO_ALMACENAMIENTO AS
BEGIN
    INSERT INTO HISTORIAL_CRECIMIENTO_BD (BYTES_DATAFILES, BYTES_TEMPFILES, BYTES_REDO, BYTES_USADOS, BYTES_LIBRES)
    SELECT
        (SELECT NVL(SUM(BYTES), 0) FROM DBA_DATA_FILES),
        (SELECT NVL(SUM(BYTES), 0) FROM DBA_TEMP_FILES),
        (SELECT NVL(SUM(BYTES * MEMBERS), NVL(SUM(BYTES), 0)) FROM V$LOG),
        (SELECT NVL(SUM(BYTES), 0) FROM DBA_SEGMENTS),
        (SELECT NVL(SUM(BYTES), 0) FROM DBA_FREE_SPACE)
    FROM DUAL;
    COMMIT;
END;
/

-- 3. Programación automática diaria con DBMS_SCHEDULER
BEGIN
    DBMS_SCHEDULER.CREATE_JOB (
        job_name        => 'JOB_HISTORIAL_ALMACENAMIENTO',
        job_type        => 'PLSQL_BLOCK',
        job_action      => 'BEGIN REGISTRAR_CRECIMIENTO_ALMACENAMIENTO; END;',
        start_date      => SYSTIMESTAMP,
        repeat_interval => 'FREQ=DAILY; BYHOUR=0; BYMINUTE=0; BYSECOND=0',
        enabled         => TRUE,
        comments        => 'Captura diaria del crecimiento de la base de datos'
    );
END;
